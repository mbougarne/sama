## Start the synthetic app (terminal A, repository root)

Copy this entire block. It serves the built UI and memory-only fixture data on
loopback. All POSTs below change fixture dictionaries only. It is **not** the Go
API and cannot establish server authorization, real authentication or provider
acceptance. Production mutation registration stays disabled.

```sh
python3 - <<'PY'
from http.server import BaseHTTPRequestHandler, HTTPServer
from pathlib import Path
from urllib.parse import urlsplit, parse_qs
from datetime import datetime, timedelta, timezone
import json

assets = (Path.cwd() / 'frontend/dist').resolve()
assert (assets / 'index.html').is_file(), 'Build frontend first from repository root'
user = '01900000-0000-7000-8000-000000000001'
workspace = '01900000-0000-7000-8000-000000000003'
operation = '01900000-0000-7000-8000-000000000005'
review = '01900000-0000-7000-8000-000000000006'
stamp = '2026-10-15T12:00:00Z'
fixture = dict(state='unknown', role='owner', released=False, lose=False, keys=[])
actions = ['compute.server.' + name for name in ['power_on', 'shutdown', 'power_off', 'reboot']]

def resource():
    return dict(id=user, connection_id=user, provider='hetzner', api_family='hetzner.cloud',
        account_identity='Synthetic local project', type='compute.server', scope='global',
        native_id='42', name='demo-web-01', region='fsn1', status='running',
        native_status='running', observed_at=stamp, details_version=1,
        details=dict(machine_type='fixture', region='fsn1'),
        inventory=dict(state='fresh', last_success_at=stamp, failure_code=None),
        available_actions=[dict(action=a, available=fixture['role'] != 'viewer',
            reason='denied' if fixture['role'] == 'viewer' else 'available') for a in actions])

def evidence():
    return dict(id=operation, resource_id=user, connection_id=user, actor_id=user,
        action=actions[3], state=fixture['state'], uncertain=fixture['state'] == 'unknown',
        reservation_released=fixture['released'], target_name='demo-web-01', native_id='42',
        provider='hetzner', account_identity='Synthetic local project', region='fsn1',
        provider_reference='fixture-action-42', created_at=stamp, updated_at=stamp,
        events=[dict(sequence=1, state=fixture['state'], created_at=stamp)])

class Fixture(BaseHTTPRequestHandler):
    def log_message(self, *args):
        pass

    def reply(self, data, status=200):
        encoded = json.dumps(data).encode()
        self.send_response(status)
        self.send_header('Content-Type', 'application/json')
        self.send_header('Cache-Control', 'no-store')
        self.send_header('Content-Length', str(len(encoded)))
        self.end_headers()
        self.wfile.write(encoded)

    def do_GET(self):
        path = urlsplit(self.path).path
        query = parse_qs(urlsplit(self.path).query)
        if path == '/_fixture':
            for field, allowed in [('state', ['queued', 'dispatching', 'waiting_provider', 'reconciling', 'unknown', 'succeeded', 'failed', 'cancelled']), ('role', ['owner', 'admin', 'operator', 'viewer'])]:
                if query.get(field, [''])[0] in allowed:
                    fixture[field] = query[field][0]
            if 'reset' in query:
                fixture.update(released=False, keys=[])
            fixture['lose'] = query.get('lose', ['0'])[0] == '1'
            return self.reply(fixture)
        if path == '/api/v1/me':
            return self.reply(dict(id=user, display_name='Synthetic local operator'))
        if path == '/api/v1/workspaces':
            return self.reply(dict(data=[dict(id=workspace, name='Local fixture', role=fixture['role'])], next_cursor=None))
        if path.endswith('/grants'):
            return self.reply(dict(actions=['read'] if fixture['role'] == 'viewer' else ['read', 'operate', 'high_impact']))
        if path.startswith('/api/') and path.endswith('/resources/' + user):
            return self.reply(resource())
        if path.startswith('/api/') and path.endswith('/operations/' + operation):
            return self.reply(evidence())
        if path.startswith('/api/') and path.endswith('/operations'):
            return self.reply(dict(data=[] if 'cursor' in query else [evidence()], next_cursor=None if 'cursor' in query else 'page2'))
        if path.startswith('/api/'):
            return self.reply(dict(code='not_found'), 404)
        relative = 'index.html' if path.startswith('/workspaces/') or path == '/' else path.lstrip('/')
        target = (assets / relative).resolve()
        if assets not in target.parents or not target.is_file():
            return self.reply(dict(code='not_found'), 404)
        data = target.read_bytes()
        self.send_response(200)
        self.send_header('Content-Type', {'.html': 'text/html', '.js': 'text/javascript', '.css': 'text/css'}.get(target.suffix, 'application/octet-stream'))
        self.send_header('Content-Length', str(len(data)))
        self.end_headers()
        self.wfile.write(data)

    def do_POST(self):
        data = json.loads(self.rfile.read(min(int(self.headers.get('Content-Length', 0)), 4096)) or '{}')
        path = urlsplit(self.path).path
        if path.endswith('/action-reviews'):
            expires = (datetime.now(timezone.utc) + timedelta(minutes=5)).isoformat()
            return self.reply(dict(id=review, resource_id=user, action=data['action'], target_name='demo-web-01', high_impact=True, expires_at=expires))
        if path.endswith('/operations'):
            key = self.headers.get('Idempotency-Key')
            if key not in fixture['keys']:
                fixture['keys'].append(key)
            fixture['state'] = 'queued'
            if fixture['lose']:
                fixture['lose'] = False
                self.close_connection = True
                return
        elif path.endswith('/cancel'):
            fixture['state'] = 'cancelled'
        elif path.endswith('/reconciliations'):
            fixture['state'] = 'reconciling'
        elif path.endswith('/reservation-release'):
            fixture['released'] = True
        return self.reply(dict(id=operation, state=fixture['state']))

print('Synthetic UI only: http://127.0.0.1:18112 — Ctrl-C stops this server')
HTTPServer(('127.0.0.1', 18112), Fixture).serve_forever()
PY
```
